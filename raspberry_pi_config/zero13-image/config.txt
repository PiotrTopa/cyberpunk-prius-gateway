# Prius Gen2 MFD video board — Raspberry Pi Zero v1.3 (32-bit), built 2026-10-14
# Carried over verbatim from the Zero 2 W card (Debian 13 image, config.txt as
# running on 2026-10-14) except: arm_64bit removed (ARMv6 board), vc4-kms-v3d
# NOT loaded (the app writes to the firmware framebuffer /dev/fb0 = BCM2708 FB).
# Reference: cyberpank-prius-gen2-computer/docs/MFD_VIDEO_TIMINGS.md

[all]
# --- SYSTEM PERFORMANCE & BOOT ---
arm_boost=1
disable_splash=1
disable_fw_kms_setup=1
max_framebuffers=2
auto_initramfs=1

# --- HARDWARE INTERFACES ---
dtparam=audio=on

# --- VGA666 CONFIGURATION ---
dtoverlay=vga666
enable_dpi_lcd=1
display_default_lcd=1

# --- DPI SETTINGS ---
# dpi_output_format=0x05 = RGB666 config 1 — the VGA666 board is wired for
# config 1. 0x06 (config 2) scrambles the bit lanes: gray ramp turns into
# saturated green/blue/pink (the "green cast", fixed 2026-09-08). Never 0x06.
dpi_group=2
dpi_mode=87
dpi_output_format=0x05

# --- CUSTOM TIMINGS (15 kHz for Prius MFD, 480x240@60.07 Hz, 9.6 MHz pixel clock) ---
dpi_timings=480 1 24 46 60 240 1 4 3 15 0 0 0 60 0 9600000 1

# --- NETWORK / USB GADGET SETTINGS ---
dtoverlay=dwc2,dr_mode=peripheral
enable_uart=1
